# --- 옛 블로그 주소는 /blog 아래로 영구 이동(301)(쿼리 문자열 유지)
GET /@alice/my-post                    301 redirect:/blog/@alice/my-post
GET /alice                             301 redirect:/blog/alice
GET /tags?tag=java                     301 redirect:/blog/tags?tag=java
GET /search?q=a%20b                    301 redirect:/blog/search?q=a%20b
GET /write                             301 redirect:/blog/write
GET /developers                        301 redirect:/blog/developers
# --- 블로그는 /blog 에서
GET /blog                              302 redirect:/blog/

GET /blog/api/v1/users/me              200 doro-blog-backend   /api/v1/users/me

GET /api/v1/auth/session               200 auth-api

# --- 도로 파티 로그인은 BFF(doro-bff)가 처리한다(브라우저가 보는 주소는 그대로, BFF 의 /bff/party/* 로 이어진다. 쿼리 문자열 유지)
GET /party/api/v1/bff/login?return=%2Fmaps%2F1          200 doro-bff-api   /bff/party/login?return=%2Fmaps%2F1
GET /party/api/v1/bff/callback?code=x&state=y           200 doro-bff-api   /bff/party/callback?code=x&state=y
POST /party/api/v1/bff/logout                           200 doro-bff-api   /bff/party/logout
# --- 세션 조회와 나머지 파티 API 는 여전히 파티 백엔드(auth_request 를 거쳐 Bearer 가 붙는다)
GET /party/api/v1/bff/session                           200 doro-party-api /api/v1/bff/session
GET /party/api/v1/maps                                  200 doro-party-api /api/v1/maps
# --- auth_request 대상은 내부 전용: 밖에서는 부를 수 없다(404)
GET /_bff_auth_party                                    404 none
